SCRIPTS Y COMANDOS (para Replit / Node):
1) instalar deps (si hace falta)
   npm init -y
   npm i fs-extra gray-matter

2) scripts/generateBlogFromJSON.js (básico)
   // node script que lee src/data/blog-ideas.json y crea/actualiza src/content/blog/{slug}.mdx
   const fs = require('fs-extra');
   const path = require('path');
   const posts = require('../src/data/blog-ideas.json');
   const outDir = path.join(__dirname,'../src/content/blog');
   fs.ensureDirSync(outDir);
   posts.forEach(p => {
     const fm = `---\ntitle: "${p.title}"\ndate: "${new Date().toISOString()}"\nseoTitle: "${p.seoTitle}"\ndescription: "${p.metaDescription}"\nkeywords: ["${p.keyword}"]\nogImage: "${p.ogImage}"\nwordcount: ${p.wordcount}\ncanonical: "/blog/${p.slug}"\nauthor: "Chroma Travel Boutique"\n---\n\n# ${p.title}\n\n> Resumen: ${p.metaDescription}\n\n## Introducción\n\n(Escribe aquí la introducción...)\n\n## Qué encontrarás en esta guía\n- Punto 1\n- Punto 2\n\n## H2 - Recomendación principal\n\n## H2 - Dónde alojarte (hoteles recomendados)\n\n## H2 - Experiencias VIP sugeridas\n\n## FAQ\n${p.faqs ? p.faqs.map(f=>`### ${f.q}\n\n${f.a}\n`).join('\\n') : ''}\n\n<!-- CTA -->\n<div className='cta'>${p.cta} — [Solicitar cotización](/contact)</div>\n`;
     fs.writeFileSync(path.join(outDir, `${p.slug}.mdx`), fm, 'utf8');
   });
   console.log('MDX files generated.');
3) Ejecutar:
   node scripts/generateBlogFromJSON.js
